function frontendUrl(path = "/"): string {
  const base =
    process.env.PANTHEON_FE_BASE_URL ||
    process.env.FRONTEND_BASE_URL ||
    process.env.PLAYWRIGHT_BASE_URL ||
    "http://127.0.0.1:5173";
  return `${base.replace(/\/$/, "")}${path}`;
}

function corsHeaders(route: Route): Record<string, string> {
  const origin = route.request().headers()["origin"] ?? "*";
  return {
    "Access-Control-Allow-Credentials": "true",
    "Access-Control-Allow-Headers":
      "accept,authorization,content-type,idempotency-key,if-match,x-bff-api-version,x-correlation-id,x-locale,x-request-id,x-tenant-id,x-trace-id",
    "Access-Control-Allow-Methods": "GET,POST,PATCH,PUT,DELETE,OPTIONS",
    "Access-Control-Allow-Origin": origin,
    "Access-Control-Expose-Headers": "x-bff-api-version,x-correlation-id,x-request-id",
  };
}

async function fulfillJson(route: Route, body: unknown, status = 200): Promise<void> {
  await route.fulfill({ body: JSON.stringify(body), contentType: "application/json", headers: corsHeaders(route), status });
}

async function install(page: Page, writes: string[]): Promise<void> {
  await page.route(/^https?:\/\/[^/]+\/(?:bff|health|healthz|readyz).*/, async (route) => {
    const request = route.request();
    const url = new URL(request.url());
    const path = url.pathname;
    if (request.method() === "OPTIONS") return route.fulfill({ headers: corsHeaders(route), status: 204 });
    if (!["GET", "HEAD"].includes(request.method()) && path.includes("alpha-factory")) writes.push(`${request.method()} ${path}`);
    if (path === "/bff/events/stream") return route.fulfill({ body: ": connected\n\n", contentType: "text/event-stream", headers: corsHeaders(route), status: 200 });
    if (path === "/health" || path === "/healthz" || path === "/readyz") return fulfillJson(route, { status: "ok", live: true, ready: true });
    if (path === "/bff/me") {
      return fulfillJson(route, { data: { environment: { name: "playwright", strict_auth: false }, tenant_id: "pantheon-dev", user: { id: "op-1", roles: ["operator", "reviewer", "approver"] } } });
    }
    if (path === "/bff/alpha-factory") return fulfillJson(route, board);
    return fulfillJson(route, { data: { items: [] }, items: [], meta: { status: "ok" }, page_info: { page_size: 0, total: 0 } });
  });
  await installOidcDevLogin(page, { goto: false, roles: ["operator", "reviewer", "approver"], tenantId: "pantheon-dev", token: LOCAL_FIXTURE_AUTH_TOKEN });
}

/** Every card's content fits inside the card, and nothing on the page sits under the assistant button. */
async function layoutProblems(page: Page): Promise<string[]> {
  return page.evaluate(() => {
    const problems: string[] = [];
    const cards = Array.from(document.querySelectorAll<HTMLElement>("[data-af-card]")).filter((el) => el.offsetParent !== null);
    if (cards.length === 0) problems.push("no visible cards");
    for (const card of cards) {
      const box = card.getBoundingClientRect();
      if (card.scrollWidth > card.clientWidth + 1) problems.push(`overflow ${card.dataset.testid} ${card.scrollWidth}>${card.clientWidth}`);
      for (const child of Array.from(card.querySelectorAll<HTMLElement>("*"))) {
        const r = child.getBoundingClientRect();
        if (r.width > 0 && (r.right > box.right + 1 || r.left < box.left - 1)) {
          problems.push(`child outside ${card.dataset.testid}: ${child.tagName} ${(child.textContent ?? "").slice(0, 40)}`);
          break;
        }
      }
    }
    if (document.documentElement.scrollWidth > window.innerWidth + 1) problems.push(`page scrolls sideways ${document.documentElement.scrollWidth}>${window.innerWidth}`);
    const button = document.querySelector<HTMLElement>('button[aria-label="開啟 Management AI"], button[title^="Management AI"]');
    if (button) {
      // Scroll every scroll container (the app shell scrolls an inner element) to its end:
      // content under the button mid-scroll is reachable; at the end nothing may stay under it.
      window.scrollTo(0, document.documentElement.scrollHeight);
      let scrolled = window.scrollY > 0;
      for (const el of Array.from(document.querySelectorAll<HTMLElement>("*"))) {
        const style = getComputedStyle(el);
        if (/(auto|scroll)/.test(style.overflowY) && el.scrollHeight > el.clientHeight + 1) {
          el.scrollTop = el.scrollHeight;
          if (el.scrollTop > 0) scrolled = true;
        }
      }
      // The fixture page is taller than both viewports, so a check that never scrolled proves nothing.
      if (!scrolled) problems.push("page did not scroll to its end");
      const b = button.getBoundingClientRect();
      const content = Array.from(document.querySelectorAll<HTMLElement>('[data-testid^="alpha-factory-"]'));
      for (const el of content) {
        const r = el.getBoundingClientRect();
        if (r.width === 0 || r.height === 0 || el.contains(button)) continue;
        // Only leaf-ish blocks matter; a lane container may legitimately span the button's column.
        if (el.children.length > 0 && el.querySelector('[data-testid^="alpha-factory-"]')) continue;
        const overlaps = r.left < b.right && r.right > b.left && r.top < b.bottom && r.bottom > b.top;
        if (overlaps) problems.push(`under assistant button: ${el.dataset.testid}`);
      }
    } else {
      problems.push("assistant button not found");
    }
    return problems;
  });
}

test.describe("FE-ALPHA-FACTORY-PIPELINE-TRUTH-20261125", () => {
  test("alpha factory shows the real pipeline at 1280 px and 390 px", async ({ page }) => {
    test.skip(targetsExternalE2eEnvironment(), "route-mocked fixture coverage is loopback-only");
    mkdirSync(EVIDENCE_DIR, { recursive: true });
    const writes: string[] = [];
    await install(page, writes);

    for (const width of [1280, 390]) {
      await page.setViewportSize({ width, height: width === 390 ? 844 : 900 });
      await page.goto(frontendUrl("/management/alpha-factory"), { waitUntil: "domcontentloaded", timeout: 30_000 });
      await expect(page.getByTestId("alpha-factory-stage-counts")).toBeVisible();
      await expect(page.getByTestId("alpha-factory-stage-passes-value")).toHaveText("0");
      await expect(page.getByTestId("alpha-factory-stage-collected-value")).toContainText("無法取得");
      await expect(page.getByTestId("alpha-factory-empty-replicated")).toHaveText("目前沒有任何候選通過回測驗證。");
      await expect(page.getByTestId("alpha-factory-item-title-rd-20261120-hengyue-effective-bets")).toContainText("衡岳 需要：");
      await expect(page.getByTestId("alpha-factory-source-banner")).not.toContainText("alpha_factory");
      await page.waitForTimeout(300);
      expect(await layoutProblems(page), `layout at ${width}px`).toEqual([]);
      // layoutProblems leaves every scroll container at its end: this shot shows the page bottom.
      await page.screenshot({ path: `${EVIDENCE_DIR}/alpha-factory-${width}w-bottom.png` });
      await page.evaluate(() => {
        window.scrollTo(0, 0);
        for (const el of Array.from(document.querySelectorAll<HTMLElement>("*"))) if (el.scrollTop > 0) el.scrollTop = 0;
      });
      await page.screenshot({ path: `${EVIDENCE_DIR}/alpha-factory-${width}w.png`, });

      await page.getByTestId("alpha-factory-document-checked").locator("summary").click();
      await expect(page.getByTestId("alpha-factory-document-checked")).toHaveAttribute("open", "");
      await page.waitForTimeout(200);
      expect(await layoutProblems(page), `layout with document checks open at ${width}px`).toEqual([]);
      await page.screenshot({ path: `${EVIDENCE_DIR}/alpha-factory-${width}w-document-checks-open-bottom.png` });
      await page.evaluate(() => {
        window.scrollTo(0, 0);
        for (const el of Array.from(document.querySelectorAll<HTMLElement>("*"))) if (el.scrollTop > 0) el.scrollTop = 0;
      });
    }
    expect(writes).toEqual([]);
  });
});
